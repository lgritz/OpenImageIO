set (OIIO_IV_EXTRA_IV_LIBRARIES "" CACHE STRING "Paths to extra libraries to force iv to link against")

check_is_enabled (iv iv_enabled)

# Sources shared by both GUI backends
set (iv_srcs imageviewer.cpp ivgl.cpp ivgl_ocio.cpp ivimage.cpp ivmain.cpp)

if (NOT iv_enabled)
    message (STATUS "${ColorRed}Disabling iv ${ColorReset}")
elseif (OIIO_IV_BACKEND STREQUAL "Qt" AND (Qt5_FOUND OR Qt6_FOUND)
        AND OPENGL_FOUND)
    message (STATUS "iv backend: Qt")
    set (CMAKE_AUTOMOC ON)
    set (CMAKE_AUTOMOC_PATH_PREFIX OFF)
    if (Qt5_POSITION_INDEPENDENT_CODE OR Qt6_POSITION_INDEPENDENT_CODE)
        set (CMAKE_POSITION_INDEPENDENT_CODE ON)
    endif()
    fancy_add_executable (
        SRC
            ${iv_srcs} ivinfowin.cpp ivpref.cpp ivui_qt.cpp
        SYSTEM_INCLUDE_DIRS
            ${OPENGL_INCLUDE_DIR}
        DEFINITIONS
            IV_USE_QT
        LINK_LIBRARIES
            OpenImageIO
            $<TARGET_NAME_IF_EXISTS:Qt5::Core>
            $<TARGET_NAME_IF_EXISTS:Qt5::Gui>
            $<TARGET_NAME_IF_EXISTS:Qt5::Widgets>
            $<TARGET_NAME_IF_EXISTS:Qt5::OpenGL>
            $<TARGET_NAME_IF_EXISTS:Qt6::Core>
            $<TARGET_NAME_IF_EXISTS:Qt6::Gui>
            $<TARGET_NAME_IF_EXISTS:Qt6::Widgets>
            $<TARGET_NAME_IF_EXISTS:Qt6::OpenGLWidgets>
            ${OPENGL_LIBRARIES}
            $<TARGET_NAME_IF_EXISTS:OpenColorIO::OpenColorIO>
            $<TARGET_NAME_IF_EXISTS:OpenColorIO::OpenColorIOHeaders>
            ${OIIO_IV_EXTRA_IV_LIBRARIES}
        )
    if (Qt5_FOUND)
        target_compile_definitions(iv PRIVATE OIIO_QT_MAJOR=5)
    endif ()
    if (Qt6_FOUND)
        target_compile_definitions(iv PRIVATE OIIO_QT_MAJOR=6)
    endif ()
elseif (OIIO_IV_BACKEND STREQUAL "ImGui" AND TARGET glfw AND OPENGL_FOUND
        AND (NOT OIIO_IV_EXTERNAL_IMGUI OR TARGET imgui::imgui))
    message (STATUS "iv backend: ImGui")

    # Third-party code bundled in ext/. Compile it without our warnings.
    set (iv_ext_srcs ext/tinyfiledialogs/tinyfiledialogs.c)
    set (iv_ext_includes ext/tinyfiledialogs ext/glad)
    if (NOT OIIO_IV_EXTERNAL_IMGUI)
        list (APPEND iv_ext_srcs
              ext/imgui/imgui.cpp
              ext/imgui/imgui_draw.cpp
              ext/imgui/imgui_tables.cpp
              ext/imgui/imgui_widgets.cpp
              ext/imgui/backends/imgui_impl_glfw.cpp
              ext/imgui/backends/imgui_impl_opengl2.cpp)
        list (APPEND iv_ext_includes ext/imgui ext/imgui/backends)
    endif ()
    if (MSVC)
        set_source_files_properties (${iv_ext_srcs} PROPERTIES COMPILE_OPTIONS "/w")
    else ()
        set_source_files_properties (${iv_ext_srcs} PROPERTIES COMPILE_OPTIONS "-w")
    endif ()

    # Embed the GUI fonts in the executable: generate a source file holding
    # the bytes of each font file.
    set (iv_fonts_cpp "${CMAKE_CURRENT_BINARY_DIR}/ivfonts.cpp")
    set (iv_fonts_text "// Generated by src/iv/CMakeLists.txt. Do not edit.\n")
    foreach (_font sans:Droid_Sans/DroidSans.ttf
                   mono:Droid_Sans_Mono/DroidSansMono.ttf)
        string (REPLACE ":" ";" _font "${_font}")
        list (GET _font 0 _name)
        list (GET _font 1 _file)
        set (_file "${PROJECT_SOURCE_DIR}/src/fonts/${_file}")
        set_property (DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_file}")
        file (READ "${_file}" _hex HEX)
        string (REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
        string (APPEND iv_fonts_text
                "extern const unsigned char iv_font_${_name}[] = {${_hex}};\n"
                "extern const unsigned int iv_font_${_name}_size = sizeof(iv_font_${_name});\n")
    endforeach ()
    file (CONFIGURE OUTPUT "${iv_fonts_cpp}" CONTENT "${iv_fonts_text}" @ONLY)

    fancy_add_executable (
        SRC
            ${iv_srcs} ivui_imgui.cpp ${iv_ext_srcs} ${iv_fonts_cpp}
        SYSTEM_INCLUDE_DIRS
            ${OPENGL_INCLUDE_DIR} ${iv_ext_includes}
        DEFINITIONS
            IV_USE_IMGUI
        LINK_LIBRARIES
            OpenImageIO
            glfw
            $<TARGET_NAME_IF_EXISTS:imgui::imgui>
            ${OPENGL_LIBRARIES}
            $<TARGET_NAME_IF_EXISTS:OpenColorIO::OpenColorIO>
            $<TARGET_NAME_IF_EXISTS:OpenColorIO::OpenColorIOHeaders>
            ${OIIO_IV_EXTRA_IV_LIBRARIES}
        )
    if (WIN32)
        target_link_libraries (iv PRIVATE comdlg32 ole32)
    endif ()
else ()
    set (_iv_needs "Qt")
    if (OIIO_IV_BACKEND STREQUAL "ImGui")
        set (_iv_needs "GLFW (glfw3)")
        if (OIIO_IV_EXTERNAL_IMGUI)
            string (APPEND _iv_needs ", imgui")
        endif ()
    endif ()
    message (STATUS "\n\n   WARNING: ${_iv_needs} or OpenGL not found -- 'iv' will not be built!\n")
endif ()

if (TARGET iv AND APPLE)
    target_compile_definitions(iv PRIVATE GL_SILENCE_DEPRECATION)
endif ()
